1"""The calibration engine boundary. 

2 

3Deliberately narrow: an engine takes reference spectra plus a recipe and hands 

4back something that can correct a spectrum's axis. Callers never learn *how* 

5the correction was derived, which is what lets a closed engine sit behind this 

6interface without leaking its method into the app, the UI, or the NeXus output.

104class CalibrationError(RuntimeError): 

105 """Raised when a recipe cannot be fitted from the given inputs. 

106 

107 ``detail`` carries the underlying traceback for an expander in the UI. The 

108 message itself should say which step failed and what the user can change -- 

109 a bare library exception leaves someone staring at a screen with no idea 

110 which of their uploads caused it. 

111 """ 

112 

113 def __init__(self, message: str, detail: str | None = None): 

114 super().__init__(message) 

115 self.detail = detail 

116

118#: Recognisable library failures, and what the user can actually do about them. 

119#: Matching on message text is brittle, but the alternative is showing a raw 

120#: scipy error to a spectroscopist, which helps nobody. 

121_HINTS: tuple[tuple[str, str], ...] = ( 

122 ( 

123 "must not exceed func output vector length", 

124 "peak finding grouped peaks so tightly that a group covers fewer data " 

125 "points than the model has parameters — typically two candidates a few " 

126 "points apart. Raise **minimum width** under Peak finding: it discards " 

127 "the narrow candidates that form these groups. Raising the prominence " 

128 "or trying another peak-finding strategy also helps. Widening the peak " 

129 "window does not, and neither does turning off peak fitting — the " 

130 "group is built and measured either way.", 

131 ), 

132 ( 

133 "No peaks found", 

134 "no peaks were found in this spectrum. Check the crop range and the " 

135 "axis units, and that the file really is the material this step " 

136 "expects.", 

137 ), 

138 ( 

139 "x-axes of the spectra should be equal", 

140 "the acquisitions are on different x axes and cannot be combined. " 

141 "They must come from the same instrument settings.", 

142 ), 

143 ( 

144 "Unsupported conversion", 

145 "the axis units do not match what this step expects. Check the units " 

146 "selected for the uploaded spectrum.", 

147 ), 

148) 

149 

150 

151def explain(error: Exception) -> str | None: 

152 """A usable suggestion for a known failure, or None.""" 

153 text = str(error) 

154 for needle, hint in _HINTS: 

155 if needle.lower() in text.lower(): 

156 return hint 

157 return None

160def merged_params( 

161 step_params: Mapping[str, Any] | None, 

162 overrides: Mapping[str, Any] | None, 

163 step_id: str, 

164) -> dict[str, Any]: 

165 """Recipe defaults overlaid with user overrides. 

166 

167 Overrides are keyed either globally (``{"should_fit": True}``) or per step 

168 (``{"x_curve": {"should_fit": True}}``); the per-step form wins. 

169 """ 

170 params = dict(step_params or {}) 

171 if not overrides: 

172 return params 

173 # A Mapping value is a per-step scope; anything else is a global override. 

174 params.update( 

175 {k: v for k, v in overrides.items() if not isinstance(v, Mapping)}, 

176 ) 

177 scoped = overrides.get(step_id) 

178 if isinstance(scoped, Mapping): 

179 params.update(scoped) 

180 return params
