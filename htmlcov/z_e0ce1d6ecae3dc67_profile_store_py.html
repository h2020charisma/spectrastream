1"""Keeps AppState.library and browser storage in step. 

2 

3Mounted once per run from the entry point, so every page sees a hydrated 

4library. The ordering rule that matters: never write until the browser has 

5answered at least once, otherwise the first render would persist an empty 

6library over the user's real profiles. 

7""" 

8 

9import streamlit as st 

10 

11from spectrastream.profiles import ProfileLibrary 

12from ui import localstore 

13from ui.state import AppState 

14 

15_PENDING_KEY = "spectrastream_pending_write" 

16_CLEAR_KEY = "spectrastream_pending_clear" 

17 

18 

19def request_save(library: ProfileLibrary) -> None: 

20 """Queue the library to be written on the next run. 

21 

22 The write is queued rather than performed because the bridge is mounted at 

23 the top of the script: by the time a page callback edits a profile, this 

24 run's component has already been instantiated and mutating its state would 

25 raise. 

26 """ 

27 st.session_state[_PENDING_KEY] = localstore.serialise(library) 

28 st.session_state.pop(_CLEAR_KEY, None) 

29 

30 

31def request_clear() -> None: 

32 st.session_state[_CLEAR_KEY] = True 

33 st.session_state.pop(_PENDING_KEY, None) 

34 

35 

36def sync(state: AppState) -> None: 

37 """Mount the bridge and reconcile it with ``state.library``.""" 

38 pending = st.session_state.pop(_PENDING_KEY, None) 

39 clear = bool(st.session_state.pop(_CLEAR_KEY, False)) 

40 

41 result = localstore.sync(pending_write=pending, clear=clear) 

42 

43 if not result.answered: 

44 # First render: the browser has not reported yet. Leave the in-memory 

45 # library alone and wait for the rerun the component will trigger. A 

46 # queued write must survive that wait, or the save is silently dropped. 

47 if pending is not None: 

48 st.session_state[_PENDING_KEY] = pending 

49 if clear: 

50 st.session_state[_CLEAR_KEY] = True 

51 return 

52 

53 state.storage_persistent = result.available 

54 

55 if not result.available: 

56 state.library_loaded = True 

57 return 

58 

59 if clear: 

60 state.library = ProfileLibrary() 

61 state.library_loaded = True 

62 return 

63 

64 # Read the browser's copy exactly once per session. After that the 

65 # in-memory library is authoritative and writes flow outwards. Re-reading 

66 # on every run would overwrite whatever the user just edited: the storage 

67 # echo always lags a save by at least one rerun. 

68 if state.library_loaded: 

69 return 

70 

71 library, problem = localstore.load_library(result.text) 

72 if problem and not st.session_state.get("spectrastream_load_warned"): 

73 st.session_state["spectrastream_load_warned"] = True 

74 st.warning(problem, icon=":material/warning:") 

75 state.library = library 

76 state.library_loaded = True 

77 

78 

79def save(state: AppState) -> None: 

80 """Persist the current library and rerun so the write actually happens.""" 

81 request_save(state.library) 

82 st.rerun()
